from dataclasses import dataclass

USUARIO, ADMIN = 1, 2


@dataclass
class Actor:
    id: int
    org_id: int
    rol: int


@dataclass
class Documento:
    org_id: int
    owner_id: int
    privado: bool


def puede_leer(doc: Documento, actor: Actor) -> bool:
    """Indica si el actor puede leer el documento."""
    if not isinstance(doc, Documento) or not isinstance(actor, Actor):
        raise ValueError("documento o actor no válido")
    if doc.org_id != actor.org_id:
        return False
    if actor.rol not in (USUARIO, ADMIN):
        return False
    if doc.privado:
        return doc.owner_id == actor.id or actor.rol == ADMIN
    return True
