from dataclasses import dataclass

USUARIO, ADMIN = 1, 2


@dataclass
class Actor:
    id: int
    org_id: int
    rol: int


@dataclass
class Documento:
    org_id: int
    owner_id: int
    privado: bool


def puede_leer(doc: Documento, actor: Actor) -> bool:
    """Indica si el actor puede leer el documento."""
    if doc is None or actor is None:
        raise ValueError("El documento y el actor no pueden ser None")
    if not isinstance(doc, Documento) or not isinstance(actor, Actor):
        raise ValueError("Entradas de tipo invalido")
    if doc.privado is None or not isinstance(doc.privado, bool):
        raise ValueError("El campo 'privado' del documento debe ser booleano")
    if not isinstance(doc.org_id, int) or not isinstance(doc.owner_id, int):
        raise ValueError("Identificadores del documento invalidos")
    if not isinstance(actor.id, int) or not isinstance(actor.org_id, int):
        raise ValueError("Identificadores del actor invalidos")
    if not isinstance(actor.rol, int) or actor.rol < USUARIO:
        raise ValueError("Rol del actor invalido")
    if doc.org_id != actor.org_id:
        raise PermissionError("El actor pertenece a otra organizacion")
    if doc.privado:
        return doc.owner_id == actor.id or actor.rol == ADMIN
    return actor.rol >= USUARIO
